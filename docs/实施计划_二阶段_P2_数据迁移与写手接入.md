### Step 3：创建 `tools/migrate_to_cloud.py`

完整内容如下，**直接写入**：

```python
"""
众生界数据迁移工具 v1.0
将本地 Qdrant 中的 collection 批量迁移到云端 ECS 上的 Qdrant。

用法：
  # 迁移所有 collection
  python tools/migrate_to_cloud.py \
      --cloud-host <ECS公网IP> --cloud-port 6333 \
      --cloud-key <云端Qdrant密码>

  # 只迁移指定 collection（断点续传）
  python tools/migrate_to_cloud.py \
      --cloud-host <ECS公网IP> --cloud-port 6333 \
      --cloud-key <云端Qdrant密码> \
      --only writing_techniques_v2,case_library_v2

注意：
  - 运行前确保 ECS 安全组已临时开放 6333 端口的公网访问
  - 迁移完成后立即关闭该入站规则（指导书第七步 7.3 节）
  - 迁移耗时取决于数据量，case_library_v2（15万条）约需 20-30 分钟
"""

import argparse
import logging
import sys
import time
from typing import Optional

from qdrant_client import QdrantClient
from qdrant_client.http.models import (
    Distance, VectorParams, SparseVectorParams,
    PointStruct, SparseIndexParams,
)

logging.basicConfig(
    level=logging.INFO,
    format="%(asctime)s [%(levelname)s] %(message)s",
    handlers=[logging.StreamHandler(sys.stdout)],
)
logger = logging.getLogger("migrate")

LOCAL_HOST = "localhost"
LOCAL_PORT = 6333
BATCH = 200   # 每批迁移条数，RTX 3070 Ti 稳定值


def get_collection_names(client: QdrantClient, only: Optional[list]) -> list[str]:
    all_cols = [c.name for c in client.get_collections().collections]
    if only:
        missing = set(only) - set(all_cols)
        if missing:
            logger.warning(f"本地不存在以下 collection，已跳过：{missing}")
        return [c for c in only if c in all_cols]
    return all_cols


def ensure_collection(src: QdrantClient, dst: QdrantClient, col_name: str) -> None:
    """在目标 Qdrant 中创建（或跳过已存在的）collection，复制源端向量配置。"""
    src_info = src.get_collection(col_name)
    vectors_config = src_info.config.params.vectors
    sparse_vectors = src_info.config.params.sparse_vectors

    existing = [c.name for c in dst.get_collections().collections]
    if col_name in existing:
        dst_info = dst.get_collection(col_name)
        dst_count = dst_info.points_count or 0
        logger.info(f"  [{col_name}] 目标已存在，当前 {dst_count} 条，继续追加...")
        return

    logger.info(f"  [{col_name}] 目标不存在，创建中...")
    kwargs = {}
    if isinstance(vectors_config, dict):
        kwargs["vectors_config"] = {
            k: VectorParams(size=v.size, distance=v.distance)
            for k, v in vectors_config.items()
        }
    else:
        kwargs["vectors_config"] = VectorParams(
            size=vectors_config.size, distance=vectors_config.distance
        )

    if sparse_vectors:
        kwargs["sparse_vectors_config"] = {
            k: SparseVectorParams(index=SparseIndexParams(on_disk=False))
            for k in sparse_vectors
        }

    dst.create_collection(collection_name=col_name, **kwargs)
    logger.info(f"  [{col_name}] 创建完成")


def migrate_collection(src: QdrantClient, dst: QdrantClient, col_name: str) -> int:
    """迁移单个 collection，返回迁移的总条数。"""
    src_total = src.get_collection(col_name).points_count or 0
    dst_start = dst.get_collection(col_name).points_count or 0
    logger.info(f"  [{col_name}] 源端 {src_total} 条 / 目标已有 {dst_start} 条")

    if dst_start >= src_total:
        logger.info(f"  [{col_name}] 数据已完整，跳过")
        return 0

    migrated = 0
    offset = None
    t0 = time.time()

    while True:
        records, next_offset = src.scroll(
            collection_name=col_name,
            limit=BATCH,
            offset=offset,
            with_payload=True,
            with_vectors=True,
        )
        if not records:
            break

        points = []
        for r in records:
            vectors = r.vector
            # qdrant_client 返回的 vector 可能是 dict（多向量）或 list（单向量）
            points.append(PointStruct(id=r.id, vector=vectors, payload=r.payload or {}))

        dst.upsert(collection_name=col_name, points=points)
        migrated += len(records)
        elapsed = time.time() - t0
        rate = migrated / elapsed if elapsed > 0 else 0
        logger.info(
            f"  [{col_name}] {migrated}/{src_total} 条"
            f"  ({rate:.0f} 条/s)"
            f"  剩余约 {(src_total - migrated) / rate:.0f}s" if rate > 0 else ""
        )

        if next_offset is None:
            break
        offset = next_offset

    dst_final = dst.get_collection(col_name).points_count or 0
    logger.info(f"  [{col_name}] 迁移完成，目标现有 {dst_final} 条")
    return migrated


def main():
    parser = argparse.ArgumentParser(description="众生界 Qdrant 数据迁移工具")
    parser.add_argument("--cloud-host", required=True, help="云端 ECS 公网 IP")
    parser.add_argument("--cloud-port", type=int, default=6333)
    parser.add_argument("--cloud-key",  required=True, help="云端 Qdrant API Key")
    parser.add_argument("--local-host", default=LOCAL_HOST)
    parser.add_argument("--local-port", type=int, default=LOCAL_PORT)
    parser.add_argument(
        "--only",
        help="只迁移指定 collection（逗号分隔），不填则迁移全部",
        default=None,
    )
    args = parser.parse_args()

    only_list = [c.strip() for c in args.only.split(",")] if args.only else None

    logger.info("=== 众生界数据迁移工具 v1.0 ===")
    logger.info(f"本地: {args.local_host}:{args.local_port}")
    logger.info(f"云端: {args.cloud_host}:{args.cloud_port}")

    src = QdrantClient(host=args.local_host, port=args.local_port)
    dst = QdrantClient(host=args.cloud_host, port=args.cloud_port, api_key=args.cloud_key)

    # 验证连接
    try:
        src.get_collections()
        logger.info("本地 Qdrant 连接正常")
    except Exception as e:
        logger.error(f"本地 Qdrant 连接失败：{e}")
        sys.exit(1)
    try:
        dst.get_collections()
        logger.info("云端 Qdrant 连接正常")
    except Exception as e:
        logger.error(f"云端 Qdrant 连接失败：{e}（请确认 ECS 安全组已开放 6333 端口）")
        sys.exit(1)

    collections = get_collection_names(src, only_list)
    logger.info(f"待迁移 collection：{collections}")

    total_migrated = 0
    for col in collections:
        logger.info(f"\n>>> 开始迁移：{col}")
        try:
            ensure_collection(src, dst, col)
            n = migrate_collection(src, dst, col)
            total_migrated += n
        except Exception as e:
            logger.error(f"  [{col}] 迁移失败：{e}")
            logger.error("  请修复后使用 --only 参数重新运行此 collection")

    logger.info(f"\n=== 迁移完成，共迁移 {total_migrated} 条 ===")
    logger.info("提醒：请立即关闭 ECS 安全组 6333 端口的公网入站规则！")


if __name__ == "__main__":
    main()
```

### Step 4：创建 `core/retrieval/cloud_client.py`

完整内容如下，**直接写入**：

```python
"""
众生界云端检索客户端 v1.0
通过阿里云 API 网关调用云端 FastAPI 检索服务。

配置来源（config.json）：
  cloud_api.endpoint   - API 网关地址，如 https://xxxxxxxx.cn-hangzhou.alicloudapi.com（API 网关分组的二级域名）
  cloud_api.appcode    - AppCode 认证码（API 网关控制台获取）
  cloud_api.enabled    - true 启用云端检索，false 退回本地

用法示例：
  from core.retrieval.cloud_client import CloudSearchClient
  client = CloudSearchClient()
  results = client.search_dense(vector=embed_result, collection="case_library_v2")
"""

import json
import logging
import os
from typing import Optional

import requests

logger = logging.getLogger(__name__)

CONFIG_PATH = os.path.join(os.path.dirname(__file__), "..", "..", "config.json")
REQUEST_TIMEOUT = 15   # 单次请求超时秒数


def _load_config() -> dict:
    with open(CONFIG_PATH, encoding="utf-8") as f:
        return json.load(f)


class CloudSearchClient:
    """阿里云 API 网关检索客户端（AppCode 认证）。"""

    def __init__(self):
        cfg = _load_config()
        cloud_cfg = cfg.get("cloud_api", {})
        self.enabled  = cloud_cfg.get("enabled", False)
        self.endpoint = cloud_cfg.get("endpoint", "").rstrip("/")
        self.appcode  = cloud_cfg.get("appcode", "")
        self._session = requests.Session()
        self._session.headers.update({
            "Authorization": f"APPCODE {self.appcode}",
            "Content-Type":  "application/json",
        })

    def _post(self, path: str, payload: dict) -> dict:
        url = f"{self.endpoint}{path}"
        try:
            resp = self._session.post(url, json=payload, timeout=REQUEST_TIMEOUT)
            resp.raise_for_status()
            return resp.json()
        except requests.exceptions.Timeout:
            raise RuntimeError(f"云端 API 超时（>{REQUEST_TIMEOUT}s）：{path}")
        except requests.exceptions.ConnectionError as e:
            raise RuntimeError(f"云端 API 连接失败：{e}")
        except requests.exceptions.HTTPError as e:
            raise RuntimeError(f"云端 API 返回错误 {resp.status_code}：{resp.text[:200]}")

    def health_check(self) -> bool:
        """检查云端服务是否正常，返回 True/False。"""
        try:
            url = f"{self.endpoint}/health"
            resp = self._session.get(url, timeout=5)
            data = resp.json()
            return data.get("status") == "ok"
        except Exception as e:
            logger.warning(f"云端健康检查失败：{e}")
            return False

    def search_dense(
        self,
        vector:          list[float],
        collection:      str,
        top_k:           int            = 5,
        score_threshold: float          = 0.0,
        filter_field:    Optional[str]  = None,
        filter_value:    Optional[str]  = None,
        vector_name:     Optional[str]  = None,
    ) -> list[dict]:
        """
        密集向量检索。

        Args:
            vector:          本地 BGE-M3 生成的查询向量（1024 维）
            collection:      目标 collection 名称
            top_k:           返回条数，1-50
            score_threshold: 最低分数阈值
            filter_field:    payload 过滤字段名（可选）
            filter_value:    payload 过滤字段值（可选）
            vector_name:     多向量 collection 时指定向量名（如 "dense"）

        Returns:
            list of {"id": ..., "score": ..., "payload": {...}}
        """
        payload = {
            "vector":           vector,
            "collection":       collection,
            "top_k":            top_k,
            "score_threshold":  score_threshold,
        }
        if filter_field:
            payload["filter_field"] = filter_field
        if filter_value:
            payload["filter_value"] = filter_value
        if vector_name:
            payload["vector_name"] = vector_name

        data = self._post("/search/dense", payload)
        logger.debug(f"dense search {collection} top_k={top_k} latency={data.get('latency_ms')}ms")
        return data.get("results", [])

    def search_hybrid(
        self,
        dense_vector:  list[float],
        collection:    str,
        sparse_vector: Optional[dict]  = None,
        top_k:         int             = 5,
        filter_field:  Optional[str]   = None,
        filter_value:  Optional[str]   = None,
    ) -> list[dict]:
        """
        混合检索（dense + sparse RRF 融合）。

        Args:
            dense_vector:  BGE-M3 dense 向量
            collection:    目标 collection（应含 dense + sparse 两条向量）
            sparse_vector: {"indices": [...], "values": [...]}（可选，不传则退化为纯 dense）
            top_k:         返回条数
            filter_field:  payload 过滤字段名（可选）
            filter_value:  payload 过滤字段值（可选）

        Returns:
            list of {"id": ..., "score": ..., "payload": {...}}
        """
        payload: dict = {
            "dense_vector": dense_vector,
            "collection":   collection,
            "top_k":        top_k,
        }
        if sparse_vector:
            payload["sparse_vector"] = sparse_vector
        if filter_field:
            payload["filter_field"] = filter_field
        if filter_value:
            payload["filter_value"] = filter_value

        data = self._post("/search/hybrid", payload)
        logger.debug(f"hybrid search {collection} top_k={top_k} latency={data.get('latency_ms')}ms")
        return data.get("results", [])
```
