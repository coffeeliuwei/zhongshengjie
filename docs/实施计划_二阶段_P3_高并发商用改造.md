"""
众生界云端检索 API 服务 v2.0（Redis 缓存 + JWT 认证）
运行在阿里云 SAE，通过 API 网关对外提供检索服务。

环境变量（SAE 控制台注入）：

  JWT_SECRET      - JWT 签名密钥（生产环境必须设置，≥32 字符随机串）

import json
import logging
import os
import time
from typing import Optional

import redis as redis_lib
from fastapi import FastAPI, HTTPException, Depends, status
from fastapi.responses import JSONResponse
from fastapi.security import HTTPBearer, HTTPAuthorizationCredentials
from jose import JWTError, jwt

QDRANT_HOST    = os.environ["QDRANT_HOST"]

JWT_SECRET     = os.environ.get("JWT_SECRET", "dev-secret-change-in-production")
JWT_ENABLED    = os.environ.get("JWT_ENABLED", "false").lower() == "true"
JWT_ALGORITHM  = "HS256"

# ── 认证 ──────────────────────────────────────────────────────────────────────

def verify_token(credentials: HTTPAuthorizationCredentials = Depends(security)):
    if not JWT_ENABLED:
        return {"sub": "anonymous"}
    if credentials is None:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="缺少 Authorization 头")
    try:
        payload = jwt.decode(credentials.credentials, JWT_SECRET, algorithms=[JWT_ALGORITHM])
        return payload
    except JWTError:
        raise HTTPException(status_code=status.HTTP_401_UNAUTHORIZED, detail="Token 无效或已过期")


# ── Redis 缓存工具 ────────────────────────────────────────────────────────────

def _cache_key(prefix: str, params: dict) -> str:
    raw = json.dumps(params, sort_keys=True, ensure_ascii=False)
    return f"{prefix}:{hashlib.md5(raw.encode()).hexdigest()}"


def _get_cache(key: str) -> Optional[dict]:
    if _redis is None:
        return None
    try:
        val = _redis.get(key)
        return json.loads(val) if val else None
    except Exception:
        return None


def _set_cache(key: str, data: dict) -> None:
    if _redis is None:
        return
    try:
        _redis.setex(key, CACHE_TTL, json.dumps(data, ensure_ascii=False))
    except Exception:
        pass

def search_dense(req: DenseSearchReq, token: dict = Depends(verify_token)):
    writer = token.get("sub", "anonymous")

    # 缓存查询（跳过 vector 本身，用摘要作 key）

        "type": "dense",
        "collection": req.collection,
        "top_k": req.top_k,
        "score_threshold": req.score_threshold,
        "filter_field": req.filter_field,

    cached = _get_cache(ck)
    if cached:
        cached["cache_hit"] = True
        return cached

    qfilter = None
    if req.filter_field and req.filter_value:
        qfilter = Filter(must=[FieldCondition(
            key=req.filter_field, match=MatchValue(value=req.filter_value)

    t0 = time.time()
    try:
        if req.vector_name:
            results = qdrant.search(
                collection_name=req.collection,
                query_vector=NamedVector(name=req.vector_name, vector=req.vector),
                limit=req.top_k, score_threshold=req.score_threshold,
                query_filter=qfilter, with_payload=True,
            )
        else:
            results = qdrant.search(
                collection_name=req.collection,
                query_vector=req.vector,
                limit=req.top_k, score_threshold=req.score_threshold,
                query_filter=qfilter, with_payload=True,

    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    latency = round((time.time() - t0) * 1000, 1)
    logger.info(f"dense | {writer} | {req.collection} top_k={req.top_k} {latency}ms")
    resp = {
        "results": [{"id": r.id, "score": r.score, "payload": r.payload} for r in results],
        "latency_ms": latency,
        "cache_hit": False,
    }
    _set_cache(ck, resp)
    return resp

        dense_res = qdrant.search(
            collection_name=req.collection,
            query_vector=NamedVector(name="dense", vector=req.dense_vector),
            limit=req.top_k * 2, query_filter=qfilter, with_payload=True,

        sparse_res = []
        if req.sparse_vector:
            sparse_res = qdrant.search(
                collection_name=req.collection,
                query_vector=NamedSparseVector(
                    name="sparse",
                    vector=SparseVector(
                        indices=req.sparse_vector.indices,
                        values=req.sparse_vector.values,

        scores:   dict = {}
        payloads: dict = {}
        for rank, r in enumerate(dense_res):
            scores[r.id]   = scores.get(r.id, 0) + 0.7 / (rank + 60)
            payloads[r.id] = r.payload
        for rank, r in enumerate(sparse_res):
            scores[r.id] = scores.get(r.id, 0) + 0.3 / (rank + 60)
            if r.id not in payloads:
                payloads[r.id] = r.payload
        fused = sorted(scores.items(), key=lambda x: -x[1])[:req.top_k]
    except Exception as e:
        raise HTTPException(status_code=500, detail=str(e))
    latency = round((time.time() - t0) * 1000, 1)
    logger.info(f"hybrid | {writer} | {req.collection} top_k={req.top_k} {latency}ms")
    resp = {
        "results": [{"id": uid, "score": s, "payload": payloads.get(uid, {})} for uid, s in fused],
        "latency_ms": latency,
        "cache_hit": False,
    }
    _set_cache(ck, resp)
    return resp

"""
三状态熔断器（CLOSED → OPEN → HALF_OPEN → CLOSED）

使用示例：
    breaker = CircuitBreaker(failure_threshold=5, recovery_timeout=60)

    @breaker
    def call_api():
        ...
"""
import logging
import time
from enum import Enum
from functools import wraps
from threading import Lock

logger = logging.getLogger(__name__)


class State(Enum):
    CLOSED    = "CLOSED"     # 正常，请求放行
    OPEN      = "OPEN"       # 熔断，请求被拒
    HALF_OPEN = "HALF_OPEN"  # 试探，放行一个请求


class CircuitBreakerOpen(Exception):
    """熔断器打开时抛出此异常。"""


class CircuitBreaker:
    """
    线程安全的三状态熔断器。

    Args:
        failure_threshold: 触发熔断的连续失败次数，默认 5
        recovery_timeout:  OPEN → HALF_OPEN 等待秒数，默认 60
    """

    def __init__(self, failure_threshold: int = 5, recovery_timeout: float = 60.0):
        self.failure_threshold = failure_threshold
        self.recovery_timeout  = recovery_timeout
        self._state            = State.CLOSED
        self._failure_count    = 0
        self._opened_at        = 0.0
        self._lock             = Lock()

    @property
    def state(self) -> State:
        return self._state

    def _transition(self, new_state: State) -> None:
        if self._state != new_state:
            logger.warning(f"熔断器状态变更：{self._state.value} → {new_state.value}")
        self._state = new_state

    def _check_state(self) -> None:
        """在请求前调用，检查是否应从 OPEN 进入 HALF_OPEN。"""
        if self._state == State.OPEN:
            if time.time() - self._opened_at >= self.recovery_timeout:
                self._transition(State.HALF_OPEN)
            else:
                raise CircuitBreakerOpen(
                    f"熔断器已打开，{self.recovery_timeout}s 后自动恢复"
                )

    def call(self, func, *args, **kwargs):
        with self._lock:
            self._check_state()
            if self._state == State.OPEN:
                raise CircuitBreakerOpen()

        try:
            result = func(*args, **kwargs)
        except Exception as e:
            with self._lock:
                if self._state == State.HALF_OPEN:
                    self._failure_count = self.failure_threshold
                    self._opened_at = time.time()
                    self._transition(State.OPEN)
                else:
                    self._failure_count += 1
                    if self._failure_count >= self.failure_threshold:
                        self._opened_at = time.time()
                        self._transition(State.OPEN)
            raise
        else:
            with self._lock:
                if self._state == State.HALF_OPEN:
                    self._failure_count = 0
                    self._transition(State.CLOSED)
                elif self._state == State.CLOSED:
                    self._failure_count = 0
            return result

    def __call__(self, func):
        @wraps(func)
        def wrapper(*args, **kwargs):
            return self.call(func, *args, **kwargs)
        return wrapper

"""
众生界云端检索客户端 v2.0（重试 + 熔断器）
通过阿里云 API 网关调用云端 FastAPI 检索服务。

配置来源（config.json）：
  cloud_api.endpoint   - API 网关地址
  cloud_api.appcode    - AppCode 认证码
  cloud_api.enabled    - true 启用云端检索
  cloud_api.jwt_token  - 写手 JWT token（由 tools/issue_token.py 生成）
"""

import json
import logging
import os
import time
from typing import Optional

import requests

from core.retrieval.circuit_breaker import CircuitBreaker, CircuitBreakerOpen

logger = logging.getLogger(__name__)

CONFIG_PATH     = os.path.join(os.path.dirname(__file__), "..", "..", "config.json")
REQUEST_TIMEOUT = 15
MAX_RETRIES     = 3
RETRY_BACKOFF   = [0.5, 1.0, 2.0]   # 每次重试等待秒数


def _load_config() -> dict:
    with open(CONFIG_PATH, encoding="utf-8") as f:
        return json.load(f)


class CloudSearchClient:
    """阿里云 API 网关检索客户端（AppCode + JWT 认证，含重试和熔断）。"""

    def __init__(self):
        cfg = _load_config()
        cloud_cfg = cfg.get("cloud_api", {})
        self.enabled   = cloud_cfg.get("enabled", False)
        self.endpoint  = cloud_cfg.get("endpoint", "").rstrip("/")
        self.appcode   = cloud_cfg.get("appcode", "")
        self.jwt_token = cloud_cfg.get("jwt_token", "")

        self._session  = requests.Session()
        self._session.headers["Authorization"] = f"APPCODE {self.appcode}"
        self._session.headers["Content-Type"]  = "application/json"
        if self.jwt_token:
            self._session.headers["X-Writer-Token"] = self.jwt_token

        self._breaker  = CircuitBreaker(failure_threshold=5, recovery_timeout=60)

    def _post_with_retry(self, path: str, payload: dict) -> dict:
        url = f"{self.endpoint}{path}"
        last_error = None
        for attempt in range(MAX_RETRIES):
            try:
                def _do_request():
                    resp = self._session.post(url, json=payload, timeout=REQUEST_TIMEOUT)
                    resp.raise_for_status()
                    return resp.json()
                return self._breaker.call(_do_request)
            except CircuitBreakerOpen as e:
                raise RuntimeError(f"云端 API 熔断器已打开：{e}") from e
            except requests.exceptions.Timeout:
                last_error = RuntimeError(f"云端 API 超时（>{REQUEST_TIMEOUT}s）")
            except requests.exceptions.ConnectionError as e:
                last_error = RuntimeError(f"云端 API 连接失败：{e}")
            except requests.exceptions.HTTPError as e:
                if hasattr(e, 'response') and e.response is not None:
                    sc = e.response.status_code
                    if sc in (400, 401, 403, 404):
                        raise RuntimeError(f"云端 API 错误 {sc}，不重试：{e.response.text[:200]}")
                last_error = RuntimeError(f"云端 API HTTP 错误：{e}")

            if attempt < MAX_RETRIES - 1:
                wait = RETRY_BACKOFF[attempt]
                logger.warning(f"请求失败（第{attempt+1}次），{wait}s 后重试：{last_error}")
                time.sleep(wait)

        raise last_error or RuntimeError("云端 API 请求失败")

    def health_check(self) -> bool:
        try:
            url = f"{self.endpoint}/health"
            resp = self._session.get(url, timeout=5)
            return resp.json().get("status") == "ok"
        except Exception as e:
            logger.warning(f"云端健康检查失败：{e}")
            return False

    def search_dense(
        self,
        vector:          list[float],
        collection:      str,
        top_k:           int           = 5,
        score_threshold: float         = 0.0,
        filter_field:    Optional[str] = None,
        filter_value:    Optional[str] = None,
        vector_name:     Optional[str] = None,
    ) -> list[dict]:
        payload = {
            "vector":          vector,
            "collection":      collection,
            "top_k":           top_k,
            "score_threshold": score_threshold,
        }
        if filter_field:  payload["filter_field"] = filter_field
        if filter_value:  payload["filter_value"] = filter_value
        if vector_name:   payload["vector_name"]  = vector_name
        data = self._post_with_retry("/search/dense", payload)
        logger.debug(f"dense {collection} top_k={top_k} latency={data.get('latency_ms')}ms cache={data.get('cache_hit')}")
        return data.get("results", [])

    def search_hybrid(
        self,
        dense_vector:  list[float],
        collection:    str,
        sparse_vector: Optional[dict] = None,
        top_k:         int            = 5,
        filter_field:  Optional[str]  = None,
        filter_value:  Optional[str]  = None,
    ) -> list[dict]:
        payload: dict = {
            "dense_vector": dense_vector,
            "collection":   collection,
            "top_k":        top_k,
        }
        if sparse_vector: payload["sparse_vector"] = sparse_vector
        if filter_field:  payload["filter_field"]  = filter_field
        if filter_value:  payload["filter_value"]  = filter_value
        data = self._post_with_retry("/search/hybrid", payload)
        logger.debug(f"hybrid {collection} top_k={top_k} latency={data.get('latency_ms')}ms cache={data.get('cache_hit')}")
        return data.get("results", [])

"""
众生界 API 压力测试脚本（Locust）

用法（在本地运行，针对已部署的 API 网关地址）：
  pip install locust
  locust -f tests/locustfile.py --host https://<API网关地址> \
         --users 50 --spawn-rate 5 --run-time 2m --headless

结果指标：
  - p95 响应时间 < 500ms 为合格
  - 错误率 < 1% 为合格
"""
import json
import random

from locust import HttpUser, task, between

# 固定测试向量（1024 维零向量，仅测吞吐，不测精度）
_DUMMY_VECTOR = [0.0] * 1024

COLLECTIONS_DENSE  = ["case_library_v2", "novel_settings_v2"]

    wait_time = between(0.5, 2.0)
    headers = {
        "Authorization": f"APPCODE {APPCODE}",

### Step 6：新建 `tools/issue_token.py`

```python
"""
众生界 JWT Token 签发工具

用法：
  # 签发 30 天有效期 token（写手 alice）
  python tools/issue_token.py --sub alice --days 30

  # 签发永不过期 token（内部测试用，不推荐生产）
  python tools/issue_token.py --sub alice --no-expiry

输出的 token 填入 config.json → cloud_api.jwt_token 字段。
"""
import argparse
import datetime
import json
import os
import sys

try:
    from jose import jwt
except ImportError:
    print("请先安装：pip install python-jose[cryptography]")
    sys.exit(1)

CONFIG_PATH = os.path.join(os.path.dirname(__file__), "..", "config.json")
ALGORITHM   = "HS256"


def load_secret() -> str:
    try:
        with open(CONFIG_PATH, encoding="utf-8") as f:
            cfg = json.load(f)
        secret = cfg.get("cloud_api", {}).get("jwt_secret", "")
        if not secret:
            raise ValueError("config.json 中 cloud_api.jwt_secret 未配置")
        return secret
    except FileNotFoundError:
        raise RuntimeError(f"config.json 不存在：{CONFIG_PATH}")


def main():
    parser = argparse.ArgumentParser(description="众生界 JWT Token 签发工具")
    parser.add_argument("--sub",       required=True, help="写手标识，如 alice")
    parser.add_argument("--days",      type=int, default=30, help="有效期天数，默认 30")
    parser.add_argument("--no-expiry", action="store_true",  help="不设过期时间")
    args = parser.parse_args()

    secret = load_secret()
    payload: dict = {"sub": args.sub, "iat": datetime.datetime.utcnow()}
    if not args.no_expiry:
        payload["exp"] = datetime.datetime.utcnow() + datetime.timedelta(days=args.days)

    token = jwt.encode(payload, secret, algorithm=ALGORITHM)
    print(f"\nJWT Token（写手：{args.sub}）：")
    print(token)
    print(f"\n有效期：{'永久' if args.no_expiry else f'{args.days} 天'}")
    print("\n将此 token 填入写手机器 config.json → cloud_api.jwt_token 字段")


if __name__ == "__main__":
    main()
```

#!/bin/bash
# 众生界 Qdrant 数据备份脚本
# 在 ECS 上运行，将 /data/qdrant_storage 打包上传到 OSS
# 建议通过 cron 每天 3:00 执行：

# 创建备份目录
mkdir -p "${BACKUP_DIR}"

# 打包（排除临时文件）
tar -czf "${ARCHIVE}" \
    --exclude="${QDRANT_DATA}/*.lock" \
    -C "$(dirname ${QDRANT_DATA})" \
    "$(basename ${QDRANT_DATA})"

SIZE=$(du -sh "${ARCHIVE}" | cut -f1)

# 上传到 OSS（需要 ossutil 已安装并配置）
ossutil cp "${ARCHIVE}" "oss://${OSS_BUCKET}/${OSS_PREFIX}/" --recursive

echo "[$(date)] 上传完成：oss://${OSS_BUCKET}/${OSS_PREFIX}/$(basename ${ARCHIVE})"

# 清理 OSS 上 30 天前的备份
EXPIRE_DATE=$(date -d "30 days ago" +%Y%m%d)
echo "[$(date)] 清理 ${EXPIRE_DATE} 之前的备份..."
ossutil ls "oss://${OSS_BUCKET}/${OSS_PREFIX}/" | \
    grep "qdrant_" | \
    awk '{print $NF}' | \
    while read obj; do
        obj_date=$(basename "${obj}" | grep -oP '\d{8}')
        if [[ "${obj_date}" < "${EXPIRE_DATE}" ]]; then
            ossutil rm "${obj}"
            echo "[$(date)] 已删除：${obj}"
        fi
    done

echo "[$(date)] 备份任务完成"

### Step 8：新建 `tools/export_billing.py`

```python
"""
众生界 API 用量计费导出工具

从阿里云 API 网关日志（或本地 SAE 日志文件）中统计各写手的调用次数。

用法：
  # 从本地日志文件统计（需先从 SLS 日志服务下载 CSV）
  python tools/export_billing.py --log-file docs/m7_artifacts/api_logs.csv \
      --output docs/m7_artifacts/billing_report.csv

日志格式（CSV，SLS 导出）：
  timestamp,writer_id,path,status_code,latency_ms
"""
import argparse
import csv
import os
import sys
from collections import defaultdict
from datetime import datetime


def parse_logs(log_file: str) -> dict:
    """解析日志 CSV，返回 {writer_id: {path: count}} 统计。"""
    stats: dict = defaultdict(lambda: defaultdict(int))
    errors: dict = defaultdict(int)

    with open(log_file, encoding="utf-8", newline="") as f:
        reader = csv.DictReader(f)
        for row in reader:
            writer = row.get("writer_id", "unknown")
            path   = row.get("path", "unknown")
            status = int(row.get("status_code", 200))
            stats[writer][path] += 1
            if status >= 400:
                errors[writer] += 1

    return {"stats": dict(stats), "errors": dict(errors)}


def export_report(data: dict, output_file: str) -> None:
    """将统计结果写入 CSV 报告。"""
    stats  = data["stats"]
    errors = data["errors"]

    with open(output_file, "w", encoding="utf-8-sig", newline="") as f:
        writer = csv.writer(f)
        writer.writerow(["写手ID", "接口", "调用次数", "错误次数", "导出时间"])
        for wid, paths in sorted(stats.items()):
            for path, count in sorted(paths.items()):
                writer.writerow([
                    wid, path, count, errors.get(wid, 0),
                    datetime.now().strftime("%Y-%m-%d %H:%M:%S"),
                ])

    print(f"报告已导出：{output_file}")
    print(f"统计写手数：{len(stats)}")
    total = sum(sum(p.values()) for p in stats.values())
    print(f"总调用次数：{total}")


def main():
    parser = argparse.ArgumentParser(description="众生界 API 计费导出工具")
    parser.add_argument("--log-file", required=True, help="SLS 导出的日志 CSV 文件路径")
    parser.add_argument("--output",   required=True, help="输出报告 CSV 路径")
    args = parser.parse_args()

    if not os.path.exists(args.log_file):
        print(f"错误：日志文件不存在：{args.log_file}", file=sys.stderr)
        sys.exit(1)

    data = parse_logs(args.log_file)
    export_report(data, args.output)


if __name__ == "__main__":
    main()
```
